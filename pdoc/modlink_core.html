181    def _run(self) -> None:
182        exit_error: Exception | None = None
183        while True:
184            try:
185                if self._drain_commands():
186                    return
187
188                frame_stream = self._frame_stream
189                if frame_stream is None:
190                    time.sleep(0.01)
191                    continue
192
193                try:
194                    frame = frame_stream.read(timeout=0.05)
195                except queue.Empty:
196                    continue
197                except StreamClosedError:
198                    if not self._started:
199                        return
200                    logger.warning("Recording frame stream closed unexpectedly; reopening stream")
201                    self._frame_stream = self._open_frame_stream()
202                    continue
203                except FrameStreamOverflowError as exc:
204                    self._handle_frame_stream_overflow(exc.consumer_name)
205                    continue
206
207                if not self.is_recording:
208                    continue
209                self._on_frame_worker(frame)
210            except Exception as exc:
211                logger.exception("Recording worker exited with an unexpected error")
212                exit_error = exc
213                break
214
215        with self._lock:
216            self._worker_exit_error = exit_error
217            self._started = False
218            self._accepting_commands = False

435    def _submit_command(
436        self,
437        worker_method: Callable[..., object],
438        *args: object,
439    ) -> Future[object]:
440        future: Future[object] = Future()
441        if not self.is_started:
442            future.set_exception(RuntimeError("ACQ_NOT_STARTED"))
443            return future
444        if not self._accepting_commands:
445            future.set_exception(RuntimeError("ACQ_SHUTTING_DOWN"))
446            return future
447
448        def _run_command() -> None:
449            try:
450                result = worker_method(*args)
451            except Exception as exc:
452                if not future.done():
453                    future.set_exception(exc)
454                return
455            if not future.done():
456                future.set_result(result)
457
458        self._command_queue.put((_run_command, future))
459        return future

322    def _tick_playback(self) -> None:
323        current_time_ns = time.monotonic_ns()
324        target_position_ns = self._sync_playback_position(current_time_ns)
325        reader = self._reader
326        if reader is None:
327            return
328        if (
329            self._timeline_index >= len(reader.frames())
330            and target_position_ns >= reader.duration_ns
331        ):
332            self._position_ns = reader.duration_ns
333            self._set_state("finished")
334            return
335        time.sleep(0.005)

337    def _sync_playback_position(self, current_time_ns: int) -> int:
338        reader = self._reader
339        if reader is None:
340            return self._position_ns
341        target_position_ns = min(
342            reader.duration_ns,
343            int(
344                self._play_started_position_ns
345                + (current_time_ns - self._play_started_wall_ns) * self._speed_multiplier
346            ),
347        )
348        self._emit_due_frames(target_position_ns)
349        self._position_ns = target_position_ns
350        return target_position_ns

352    def _emit_due_frames(self, target_position_ns: int) -> None:
353        reader = self._reader
354        if reader is None:
355            return
356        timeline = reader.frames()
357        while self._timeline_index < len(timeline):
358            ref = timeline[self._timeline_index]
359            if ref.relative_timestamp_ns > target_position_ns:
360                break
361            self.bus.ingest_frame(reader.load_frame(ref))
362            self._timeline_index += 1

365        for stream_id in tuple(self.bus.descriptors()):
366            self.bus.remove_descriptor(stream_id)
367        self.bus.add_descriptors(reader.descriptors().values())

384    ) -> Future[object]:
385        future: Future[object] = Future()
386        if not self.is_started:

394            try:
395                result = worker_method(*args)
396            except Exception as exc:
397                if not future.done():
398                    future.set_exception(exc)
399                return
400            if not future.done():
401                future.set_result(result)
402
403        self._command_queue.put((_run_command, future))
404        return future
